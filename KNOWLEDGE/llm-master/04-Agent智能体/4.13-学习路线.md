### 学 Agent 之前，先承认一个事实

Agent 是大模型应用里**最容易做出 Demo、也最容易翻车**的方向。

跟着教程，半小时就能搭一个"会调工具的 Agent"，看着很唬人。但一上真实业务，立刻死循环、乱调工具、上下文污染、权限越界……一地鸡毛。

所以学 Agent 的目标不是"搭一个能跑的 Demo"，而是**搞懂它为什么转得动、又为什么会翻车、翻车了怎么兜底**。下面这条路线就是照着这个目标排的。

### 面试高频追问

【⭐ 理解】如何向面试官证明自己真正掌握了 Agent 开发能力，而非只看过教程？
结论先行： 必须拿出“可复现的评测任务集 + 闭环故障注入排障记录 + 真实指标”。
展开解释： 调库写 Demo 门槛极低。面试官更看重：你是否能讲清 ReAct 循环的状态机转移、是否手写过基于 JSON-RPC 的工具分发、在线上遇到工具幻觉与超长死锁时如何实现双模熔断，以及是否构建过离线轨迹评测集（如包含 50 条复杂故障例，端到端通过率从 42% 提升至 88%）。

【⭐ 理解】Demo 能跑但在外企面试中被深挖底层时卡壳，如何科学建立知识体系？
结论先行： 遵循“输入→变换→输出→故障→参数”五要素，闭卷推演消息流。
展开解释： 备考切忌背代码。针对每个 Agent 组件：闭卷画出 Context 内存结构图，手撕最小 O-T-A 循环；人为注入 API 异常、脏 JSON、交替循环，验证系统的恢复与截断兜底；最后对照原仓库标准测试集验证边界，才能在技术深挖中从容应对。

【◻ 按需】自动化学习计划算法如果返回空候选列表，该如何定位故障？
结论先行： 必须先区分“全量达成”还是“图遍历死锁/环路阻塞”。
展开解释： 若总技能中尚有未掌握节点却返回空列表，必然存在死锁或依赖环路。排查时应先用入度算法检验是否有循环依赖（如 A 依赖 B，B 依赖 A），再检查是否有节点依赖了未定义的非法 ID，严禁将空结果默认当成全部通关。

### 生产级代码实现：技能拓扑树依赖推导与防环就绪断言器 (Level 13)

```python
"""4.13 Agent 技能拓扑依赖推导与防环就绪断言器 (Level 13 进阶实战)

业务场景：
在 Agent 学习与工程师能力进阶中，技能之间存在严格的先修依赖链条（如掌握单工具 ReAct 才能进阶多角色）。
本模块实现零外部依赖的轻量级拓扑排序与依赖推导引擎，提供：
1. 依赖图有向无环性 (DAG) 严格检测，防止环路死锁；
2. 技能就绪度状态机推导（必须具备复述、手撕、排障三重客观证据）；
3. 严格的非法依赖校验与异常断言，提供标准化排障诊断报告。
"""

from __future__ import annotations
from dataclasses import dataclass, field
from typing import Dict, List, Set, Tuple


@dataclass(frozen=True)
class MasteryEvidence:
    """技能掌握证据卡：杜绝'只看不练'"""
    feynman_recited: bool = False      # 能否闭卷复述核心原理与因果边界
    scratch_implemented: bool = False  # 能否纯手撕核心代码并一次跑通
    troubleshoot_proven: bool = False  # 是否具备真实错误注入与排障记录

    @property
    def is_fully_mastered(self) -> bool:
        """严苛口径：三项证据全部具备才算真正掌握"""
        return self.feynman_recited and self.scratch_implemented and self.troubleshoot_proven


@dataclass
class SkillNode:
    """技能树节点"""
    skill_id: str                      # 技能唯一标识（如 4.4_react_loop）
    title: str                         # 技能名称
    prerequisites: Set[str] = field(default_factory=set)  # 先修技能 ID 集合
    evidence: MasteryEvidence = field(default_factory=MasteryEvidence)


class SkillRoadmapEngine:
    """技能拓扑推导与就绪断言引擎"""

    def __init__(self, skills: List[SkillNode]) -> None:
        if not skills:
            raise ValueError("技能树节点列表不可为空")
        
        self.skills_map: Dict[str, SkillNode] = {s.skill_id: s for s in skills}
        if len(self.skills_map) != len(skills):
            raise ValueError("存在重复的技能 ID 定义")
            
        self._validate_integrity()

    def _validate_integrity(self) -> None:
        """校验图完整性：是否存在悬空引用与循环依赖"""
        # 1. 检查悬空依赖（引用的前置不存在）
        for skill_id, node in self.skills_map.items():
            for prereq_id in node.prerequisites:
                if prereq_id not in self.skills_map:
                    raise KeyError(f"技能 '{skill_id}' 引用了不存在的前置技能 ID: '{prereq_id}'")

        # 2. 检查循环依赖（Kahn 算法拓扑排序判环）
        in_degree: Dict[str, int] = {sid: len(node.prerequisites) for sid, node in self.skills_map.items()}
        # 依赖项到后继节点的反向映射
        successors: Dict[str, List[str]] = {sid: [] for sid in self.skills_map}
        for sid, node in self.skills_map.items():
            for pid in node.prerequisites:
                successors[pid].append(sid)

        zero_in_degree = [sid for sid, deg in in_degree.items() if deg == 0]
        visited_count = 0

        while zero_in_degree:
            curr = zero_in_degree.pop(0)
            visited_count += 1
            for nxt in successors[curr]:
                in_degree[nxt] -= 1
                if in_degree[nxt] == 0:
                    zero_in_degree.append(nxt)

        if visited_count != len(self.skills_map):
            cycle_nodes = [sid for sid, deg in in_degree.items() if deg > 0]
            raise ValueError(f"技能拓扑树存在循环依赖死锁！涉及环路节点: {cycle_nodes}")

    def get_next_actionable_skills(self) -> List[str]:
        """推导当前可立即开始练习的技能候选集
        
        触发规则：
        1. 自身尚未完全掌握 (is_fully_mastered 为 False)；
        2. 其所有的先修依赖技能均已完全掌握。
        """
        actionable: List[str] = []
        for skill_id, node in self.skills_map.items():
            if node.evidence.is_fully_mastered:
                continue  # 已通关，跳过
            
            # 检查前置是否全量就绪
            prereqs_satisfied = all(
                self.skills_map[pid].evidence.is_fully_mastered
                for pid in node.prerequisites
            )
            if prereqs_satisfied:
                actionable.append(skill_id)

        return sorted(actionable)

    def is_all_completed(self) -> bool:
        """所有技能是否全部达成全量掌握"""
        return all(node.evidence.is_fully_mastered for node in self.skills_map.values())


def self_test():
    """验证技能拓扑推导、证据门控与异常防环测试"""
    # 构建 4 个具备先后梯度的 Agent 核心技能
    skills_data = [
        SkillNode(
            skill_id="4.1_agent_structure",
            title="Agent 基本结构与状态机",
            prerequisites=set(),
            evidence=MasteryEvidence(feynman_recited=True, scratch_implemented=True, troubleshoot_proven=True)
        ),
        SkillNode(
            skill_id="4.3_tool_schema",
            title="工具 Schema 与安全沙箱",
            prerequisites={"4.1_agent_structure"},
            evidence=MasteryEvidence(feynman_recited=True, scratch_implemented=True, troubleshoot_proven=True)
        ),
        SkillNode(
            skill_id="4.4_react_loop",
            title="ReAct 工具循环与熔断内核",
            prerequisites={"4.1_agent_structure", "4.3_tool_schema"},
            # 尚未完全掌握：仅手撕代码，未做排障
            evidence=MasteryEvidence(feynman_recited=True, scratch_implemented=True, troubleshoot_proven=False)
        ),
        SkillNode(
            skill_id="4.10_multi_role",
            title="多角色协作与状态交接",
            prerequisites={"4.4_react_loop"},
            evidence=MasteryEvidence()
        ),
    ]

    engine = SkillRoadmapEngine(skills_data)

    # 1. 验证就绪推荐：4.4 的前置(4.1, 4.3)均已通关，当前唯一下一就绪目标应为 4.4
    next_tasks = engine.get_next_actionable_skills()
    assert next_tasks == ["4.4_react_loop"], f"期望推荐 4.4，实际推荐: {next_tasks}"
    # 4.10 的前置 4.4 尚未通关，绝对不可被推荐
    assert "4.10_multi_role" not in next_tasks

    # 2. 模拟通关 4.4 的排障测试，更新状态并重新推导
    engine.skills_map["4.4_react_loop"].evidence = MasteryEvidence(
        feynman_recited=True, scratch_implemented=True, troubleshoot_proven=True
    )
    next_tasks_updated = engine.get_next_actionable_skills()
    assert next_tasks_updated == ["4.10_multi_role"]

    # 3. 模拟全量通关
    engine.skills_map["4.10_multi_role"].evidence = MasteryEvidence(
        feynman_recited=True, scratch_implemented=True, troubleshoot_proven=True
    )
    assert engine.is_all_completed() is True
    assert engine.get_next_actionable_skills() == []

    # 4. 边界测试：捕获悬空不存在的前置 ID
    try:
        broken_skills = [
            SkillNode(skill_id="bad_skill", title="非法依赖", prerequisites={"non_existent_skill"})
        ]
        SkillRoadmapEngine(broken_skills)
        raise AssertionError("未捕获悬空前置依赖错误")
    except KeyError as e:
        assert "non_existent_skill" in str(e)

    # 5. 边界测试：捕获死锁依赖环路 (A -> B -> A)
    try:
        cyclic_skills = [
            SkillNode(skill_id="node_a", title="节点A", prerequisites={"node_b"}),
            SkillNode(skill_id="node_b", title="节点B", prerequisites={"node_a"}),
        ]
        SkillRoadmapEngine(cyclic_skills)
        raise AssertionError("未捕获循环依赖错误")
    except ValueError as e:
        assert "循环依赖死锁" in str(e)

    print("✅ 4.13 技能拓扑树依赖推导与防环就绪断言器自检全部通过！")


if __name__ == "__main__":
    self_test()
```

### 排障指南（常见故障、定位方法、修复策略）

| 就绪推荐为空且未通关 | get_next_actionable_skills() 为空，但 is_all_completed() 为 False。 | 检查是否存在某个未掌握的前置节点卡住了下游整条链路，优先攻克该瓶颈节点。 |

### 工程落地经验（生产防坑小贴士）

- 依赖颗粒度不可过细：切忌将单个函数甚至单个变量作为前置依赖，否则拓扑图维护成本激增；建议以“独立运行模块”为最小单元。

- 证据留存客观可溯：每次练习必须产出带单元测试的脚本与 Markdown 运行日志，将排障过程写成“故障原因-复现代码-修复方案”三段式。

- 避免闭门造车：定期挑选工业安全法规等复杂业务场景，将 RAG 检索与 Agent 工具调用组合串联，检验真实场景下的鲁棒性。

备考闪卡 Q&A：

- Q：DAG 拓扑排序检测环路的核心算法是什么？
A： Kahn 算法（基于入度统计）或深度优先搜索（DFS 染色法）。Kahn 算法中，若出队节点数小于总节点数，说明剩余节点存在环路依赖。

- Q：什么是 Agent 掌握的三重证据？
A： 概念能闭卷复述因果、内核能纯 Python 手撕、故障注入后能排障修复。
