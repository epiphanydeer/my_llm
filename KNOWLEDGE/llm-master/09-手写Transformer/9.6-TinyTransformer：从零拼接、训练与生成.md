### 一、提出问题与真实痛点

纸上谈兵终觉浅。很多初学者学完了所有算子，却从未亲手写过一个完整的自回归生成大模型循环：
- “输入一句话，大模型究竟在底层是怎么**一个词一个词蹦出来（Autoregressive Generation）**的？”
- **生产核心闭环**：
  1. 输入 Prompt 经过 Tokenizer 编码为整数列表；
  2. 构造嵌入张量并叠加绝对/旋转位置编码；
  3. 流经 $N$ 层堆叠 Transformer Block；
  4. 最终归一化后经过 LM Head 产生词表维度上的 Logits；
  5. 提取最后一个位置的 Logits，利用贪心策略或温度采样选出新的 Token ID；
  6. 将新 ID 拼回输入序列，不断自回归循环，直到生成结束符 `<|endoftext|>` 或达到最大步数！
- 本节目标：从零手写完整的 **TinyTransformer** 纯 Python/NumPy 模型，彻底打通自回归文本生成闭环！

### 三、核心概念与五要素表

| 五要素 | TinyTransformer 端到端生成模型 |
|---|---|
| **输入 (Input)** | 初始提示词 ID 列表 `prompt_tokens`，最大生成步长 `max_new_tokens`，采样温度 `temperature`。 |
| **做什么 (Process)** | 查表得到 Embedding；流经多层 Block；LM Head 输出 Logits；贪心或概率采样；自回归拼接新 Token；检测 EOS 停机。 |
| **输出 (Output)** | 生成完成的完整 Token ID 序列。 |
| **哪里出错 (Failure Modes)** | 1. 忘记加 EOS 停机检测导致死循环生成；2. 生成步骤中误取了非最后位置的 Logits。 |
| **参数影响 (Parameters)** | 温度系数 $	au$：$	au 	o 0$ 退化为贪心确定性输出；$	au > 1$ 增加多样性但增加胡言乱语风险。 |

### 四、代码动手实践与严格断言

```python
"""9.6 纯手撕端到端 TinyTransformer 与自回归生成引擎

纯 NumPy 实现：
1. 词嵌入 Embedding 表与位置编码；
2. 堆叠两层完整 Transformer Block；
3. 最终 RMSNorm 与 LM Head 输出投影；
4. 纯手写贪心与温度系数自回归生成器 generate()；
5. 严格验证多步自回归生成与单调长度增长断言。
"""
import numpy as np

def silu(x: np.ndarray) -> np.ndarray:
    return x / (1.0 + np.exp(-np.clip(x, -20.0, 20.0)))

class TinyRMSNorm:
    def __init__(self, dim: int, eps: float = 1e-6):
        self.eps = eps
        self.gamma = np.ones(dim, dtype=np.float32)

    def forward(self, x: np.ndarray) -> np.ndarray:
        rms = np.sqrt(np.mean(x ** 2, axis=-1, keepdims=True) + self.eps)
        return (x / rms) * self.gamma

class TinyTransformerBlock:
    def __init__(self, dim: int, heads: int, seed: int):
        np.random.seed(seed)
        self.dim = dim
        self.heads = heads
        self.d_k = dim // heads
        self.norm1 = TinyRMSNorm(dim)
        self.norm2 = TinyRMSNorm(dim)
        
        self.w_q = np.random.randn(dim, dim) * 0.05
        self.w_k = np.random.randn(dim, dim) * 0.05
        self.w_v = np.random.randn(dim, dim) * 0.05
        self.w_o = np.random.randn(dim, dim) * 0.05
        
        self.w_gate = np.random.randn(dim, dim * 2) * 0.05
        self.w_up = np.random.randn(dim, dim * 2) * 0.05
        self.w_down = np.random.randn(dim * 2, dim) * 0.05

    def forward(self, x: np.ndarray) -> np.ndarray:
        B, T, C = x.shape
        # MHA
        norm_x = self.norm1.forward(x)
        q = np.matmul(norm_x, self.w_q).reshape(B, T, self.heads, self.d_k).swapaxes(1, 2)
        k = np.matmul(norm_x, self.w_k).reshape(B, T, self.heads, self.d_k).swapaxes(1, 2)
        v = np.matmul(norm_x, self.w_v).reshape(B, T, self.heads, self.d_k).swapaxes(1, 2)
        
        scores = np.matmul(q, k.swapaxes(-1, -2)) / np.sqrt(self.d_k)
        mask = np.triu(np.ones((T, T), dtype=np.float32), k=1) * -1e9
        scores += mask
        
        exp_s = np.exp(scores - np.max(scores, axis=-1, keepdims=True))
        weights = exp_s / np.sum(exp_s, axis=-1, keepdims=True)
        attn_out = np.matmul(weights, v).swapaxes(1, 2).reshape(B, T, C)
        h = x + np.matmul(attn_out, self.w_o)
        
        # SwiGLU FFN
        norm_h = self.norm2.forward(h)
        gate = silu(np.matmul(norm_h, self.w_gate))
        up = np.matmul(norm_h, self.w_up)
        ffn_out = np.matmul(gate * up, self.w_down)
        return h + ffn_out

class TinyTransformerLM:
    def __init__(self, vocab_size: int = 30, embed_dim: int = 16, num_layers: int = 2, max_len: int = 32, seed: int = 42):
        self.vocab_size = vocab_size
        self.embed_dim = embed_dim
        self.max_len = max_len
        
        np.random.seed(seed)
        self.token_embeddings = np.random.randn(vocab_size, embed_dim) * 0.05
        self.pos_embeddings = np.random.randn(max_len, embed_dim) * 0.05
        
        self.blocks = [TinyTransformerBlock(embed_dim, heads=4, seed=seed + i) for i in range(num_layers)]
        self.final_norm = TinyRMSNorm(embed_dim)
        self.lm_head = np.random.randn(embed_dim, vocab_size) * 0.05

    def forward(self, input_ids: np.ndarray) -> np.ndarray:
        """input_ids: (B, T) -> Logits: (B, T, vocab_size)"""
        B, T = input_ids.shape
        assert T <= self.max_len, "输入序列超过最大长度限制！"
        
        # 查表加和
        tok_emb = self.token_embeddings[input_ids]     # (B, T, C)
        pos_emb = self.pos_embeddings[np.arange(T)]     # (T, C)
        x = tok_emb + pos_emb
        
        # 逐层前向
        for block in self.blocks:
            x = block.forward(x)
            
        x = self.final_norm.forward(x)
        logits = np.matmul(x, self.lm_head)  # (B, T, V)
        return logits

    def generate(self, prompt_tokens: list[int], max_new_tokens: int = 5, eos_token_id: int = 2) -> list[int]:
        """贪心自回归文本生成循环"""
        current_tokens = list(prompt_tokens)
        
        for _ in range(max_new_tokens):
            if len(current_tokens) >= self.max_len:
                break
            # 输入当前完整序列: (1, T)
            input_arr = np.array([current_tokens], dtype=np.int32)
            logits = self.forward(input_arr)  # (1, T, V)
            
            # 核心：只提取最后一个时间步的 Logits！
            next_token_logits = logits[0, -1, :]  # (V,)
            # 贪心选择最大概率 Token
            next_token_id = int(np.argmax(next_token_logits))
            
            current_tokens.append(next_token_id)
            if next_token_id == eos_token_id:
                break
                
        return current_tokens

def self_test():
    vocab_size = 20
    model = TinyTransformerLM(vocab_size=vocab_size, embed_dim=16, num_layers=2, max_len=16)
    
    # 模拟输入 Prompt
    prompt = [5, 8]
    generated = model.generate(prompt, max_new_tokens=4, eos_token_id=2)
    
    print(f"9.6 自检: 初始 Prompt = {prompt}, 生成后序列 = {generated}")
    # 核心不变量断言：生成后的序列必须严格包含原始 Prompt 作为前缀，且长度增加！
    assert len(generated) > len(prompt), "生成长度未增长！"
    assert generated[:len(prompt)] == prompt, "生成的 Prompt 前缀遭到篡改！"
    print("✅ 9.6 纯手写 TinyTransformer 端到端自回归生成闭环自检全部通过！")

if __name__ == '__main__':
    self_test()
```

    ```python
    def generate(model, input_ids, max_new_tokens, eos_id):
        curr_ids = list(input_ids)
        for _ in range(max_new_tokens):
            # 1. 前向传播获取整条序列 logits: (1, T, V)
            logits = model.forward(curr_ids)
            # 2. 核心：只提取最新位置的 logits 向量
            last_logits = logits[0, -1, :]
            # 3. 采样下一个词 (贪心或 Top-p)
            next_id = argmax(last_logits)
            # 4. 追加回序列
            curr_ids.append(next_id)
            # 5. 命中结束符退出
            if next_id == eos_id:
                break
        return curr_ids
    ```
